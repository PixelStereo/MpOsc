cmake_minimum_required(VERSION 3.22)
project(MP_OSC VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network Multimedia)
qt_add_executable(MP_OSC WIN32 MACOSX_BUNDLE main.cpp OscCodec.cpp SliderField.cpp SliderField.h)
set_target_properties(MP_OSC PROPERTIES AUTOMOC ON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS scripts/mediapipe_worker.py)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/scripts/mediapipe_worker.py" WORKER)
configure_file(Worker.h.in generated/Worker.h @ONLY)
target_include_directories(MP_OSC PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(MP_OSC PRIVATE Qt6::Widgets Qt6::Network Qt6::Multimedia)
set_target_properties(MP_OSC PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER com.pixelstereo.mp-osc
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
